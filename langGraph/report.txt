```python
from pathlib import Path
from typing import Optional
import pandas as pd
from pydantic import BaseSettings, FilePath

class Settings(BaseSettings):
    csv_path: FilePath = Path("Housing.csv")
    max_file_size_mb: int = 100

    class Config:
        env_file = ".env"

def load_csv(settings: Optional[Settings] = None) -> pd.DataFrame:
    """Load and validate CSV file.

    Args:
        settings: Application settings. If None, uses default settings.

    Returns:
        Loaded pandas DataFrame

    Raises:
        FileNotFoundError: If CSV file doesn't exist
        ValueError: If file is too large or invalid format
    """
    settings = settings or Settings()

    if not settings.csv_path.exists():
        raise FileNotFoundError(f"CSV file not found at {settings.csv_path}")

    if settings.csv_path.stat().st_size > settings.max_file_size_mb * 1024 * 1024:
        raise ValueError(f"File exceeds maximum size of {settings.max_file_size_mb}MB")

    try:
        return pd.read_csv(settings.csv_path)
    except Exception as e:
        raise ValueError(f"Failed to load CSV file: {str(e)}") from e
```
