# src/agents/eda_agent.py
"""Exploratory Data Analysis Agent module."""

import logging
from typing import Dict, List, Optional, Union

import pandas as pd
from pydantic import BaseModel, Field

logger = logging.getLogger(__name__)

class EDAResult(BaseModel):
    """Model for EDA results."""
    numeric_stats: Dict[str, Dict[str, Union[int, float]]] = Field(
        ..., description="Statistics for numeric columns"
    )
    categorical_stats: Dict[str, Dict[str, Union[int, str]]] = Field(
        ..., description="Statistics for categorical columns"
    )
    missing_values: Dict[str, int] = Field(
        ..., description="Count of missing values per column"
    )

class EDAAgent:
    """Agent for performing Exploratory Data Analysis."""

    def __init__(self, df: pd.DataFrame):
        """Initialize the EDA Agent with a dataframe.

        Args:
            df: Pandas DataFrame to analyze
        """
        self.df = df
        self._validate_dataframe()

    def _validate_dataframe(self) -> None:
        """Validate the input dataframe."""
        if not isinstance(self.df, pd.DataFrame):
            raise ValueError("Input must be a pandas DataFrame")
        if self.df.empty:
            raise ValueError("DataFrame cannot be empty")

    def analyze(self) -> EDAResult:
        """Perform comprehensive EDA on the dataframe.

        Returns:
            EDAResult containing analysis results
        """
        logger.info("Starting EDA analysis")

        try:
            numeric_stats = self._analyze_numeric_columns()
            categorical_stats = self._analyze_categorical_columns()
            missing_values = self._count_missing_values()

            return EDAResult(
                numeric_stats=numeric_stats,
                categorical_stats=categorical_stats,
                missing_values=missing_values
            )
        except Exception as e:
            logger.error(f"Error during EDA analysis: {str(e)}")
            raise

    def _analyze_numeric_columns(self) -> Dict[str, Dict[str, Union[int, float]]]:
        """Analyze numeric columns in the dataframe."""
        numeric_cols = self.df.select_dtypes(include=['number']).columns
        stats = {}

        for col in numeric_cols:
            stats[col] = {
                'mean': self.df[col].mean(),
                'median': self.df[col].median(),
                'std': self.df[col].std(),
                'min': self.df[col].min(),
                'max': self.df[col].max(),
                'skewness': self.df[col].skew(),
                'kurtosis': self.df[col].kurtosis()
            }

        return stats
